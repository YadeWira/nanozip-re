set(CMAKE_CXX_EXTENSIONS OFF)

    src/sfx_archive.cpp
    src/linux32_cm_map.cpp
    src/lzpf_arith.cpp
    src/nz_lzpf_encoder.cpp
    src/nz_lzpf_audio_enc.cpp
    src/nz_lzpf_image_enc.cpp
    src/nz_lzhd_encoder.cpp
    src/nz_lzhd_text.cpp
    src/nz_optimum_text.cpp
    src/nz_optimum_seg.cpp
    src/nz_optimum_param1.cpp
    src/nz_lzhds_encoder.cpp
    src/nz_cm.cpp
    src/nz_cd_tokens.cpp
    src/nz_cd_texttransform_dict.cpp
    src/nz_lzhds.cpp
    src/nz_text_transform.cpp
    src/nz_tt_chess.cpp
    src/nz_postfilter.cpp
    src/nz_bwt.cpp
    src/nz_audio.cpp
    src/nz_exefilter.cpp
    src/nz_texttransform_num.cpp
    src/nz_optimum_lz.cpp
    src/nz_optimum_lz_parser.cpp
    src/nz_optimum_lz_tables.cpp
    src/nz_optimum2_lz.cpp
    src/nz_optimum2_lz_tables.cpp
)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)

add_executable(nz-re ${RECON_SOURCES})
add_executable(nz-re-sfx ${RECON_SOURCES})

target_include_directories(nz-re PRIVATE include)
target_include_directories(nz-re-sfx PRIVATE include)

if(MSVC)
    target_compile_options(nz-re PRIVATE /W4)
    target_compile_options(nz-re-sfx PRIVATE /W4)
else()
    target_compile_options(nz-re PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(nz-re-sfx PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Optional unit-test binary for the lzpf arith primitives.
add_executable(test_lzpf_arith
    tests/test_lzpf_arith.cpp
    src/lzpf_arith.cpp
    src/nz_cd_tokens.cpp
    src/nz_cd_texttransform_dict.cpp
    src/nz_lzhds.cpp
    # nz_cd_tokens' text pipeline calls the shared tt 0x40 transform, which
    # lives in its own dependency-free unit for exactly this reason
    src/nz_tt_chess.cpp
    # ... and the tt 0x04 html transform, which the -cd pipeline dispatches too
    src/nz_text_transform.cpp
    # nz_cd_tokens' 0xf sub-chunk runs the image model (nz_audio.cpp), whose
    # bit-count model is seeded from nz_cm's shared LUT
    src/nz_audio.cpp
    src/nz_cm.cpp
)
target_include_directories(test_lzpf_arith PRIVATE include)
if(NOT MSVC)
    target_compile_options(test_lzpf_arith PRIVATE -Wall -Wextra -Wpedantic)
endif()

# The -co text detector against golden values read out of the original binary.
# The fixtures are generated by the test itself (`--dump <dir>` writes them out,
# which is how the goldens were captured), so it needs no data files.
# The detector reaches the -cd text tables, which sit in the decoder's own
# translation units, so the whole tree links in -- minus main.cpp, whose place
# the test itself takes.
set(RECON_SOURCES_NO_MAIN ${RECON_SOURCES})
list(REMOVE_ITEM RECON_SOURCES_NO_MAIN src/main.cpp)
add_executable(test_co_textdetect
    tests/test_co_textdetect.cpp
    ${RECON_SOURCES_NO_MAIN}
)
target_include_directories(test_co_textdetect PRIVATE include)
find_package(Threads REQUIRED)
target_link_libraries(test_co_textdetect PRIVATE Threads::Threads)

# The -co text transforms' forward passes against the original's side streams
# and per-step output, on the same generated fixtures.
add_executable(test_co_texttransform
    tests/test_co_texttransform.cpp
    ${RECON_SOURCES_NO_MAIN}
)
target_include_directories(test_co_texttransform PRIVATE include)
target_link_libraries(test_co_texttransform PRIVATE Threads::Threads)

# param1 (the -co AddBytes delta filter): encoder <-> decoder round trip.
add_executable(test_co_param1
    tests/test_co_param1.cpp
    ${RECON_SOURCES_NO_MAIN}
)
target_include_directories(test_co_param1 PRIVATE include)
target_link_libraries(test_co_param1 PRIVATE Threads::Threads)
